cmake_minimum_required(VERSION 3.25)
project(indago_payload_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(INDAGO_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
include("${INDAGO_ROOT}/cmake/IntegratedZstd.cmake")
add_executable(indago_payload_pack main.cpp "${INDAGO_ROOT}/src/sha256.cpp")
target_include_directories(indago_payload_pack PRIVATE "${INDAGO_ROOT}/include" "${INDAGO_ROOT}/vendor" "${INDAGO_ROOT}/vendor/zstd/upstream/lib")
target_link_libraries(indago_payload_pack PRIVATE libzstd_static)
add_executable(indago_payload_codec_tests "${INDAGO_ROOT}/tests/payload_codec_tests.cpp")
target_include_directories(indago_payload_codec_tests PRIVATE "${INDAGO_ROOT}/vendor/zstd/upstream/lib")
target_link_libraries(indago_payload_codec_tests PRIVATE libzstd_static)
enable_testing()
add_test(NAME payload_codec COMMAND indago_payload_codec_tests)
set_tests_properties(payload_codec PROPERTIES TIMEOUT 30)
