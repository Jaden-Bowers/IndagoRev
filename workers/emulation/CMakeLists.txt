cmake_minimum_required(VERSION 3.25)
project(indago_emulation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UNICORN_ARCH x86 CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(unicorn URL https://github.com/unicorn-engine/unicorn/archive/refs/tags/2.1.4.tar.gz URL_HASH SHA256=ea8863f095a0136388694e5a6063afd9bb7650e30243dd6251af59c5ce5601f4 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(unicorn)
add_executable(indago_emulation_worker main.cpp)
target_include_directories(indago_emulation_worker PRIVATE ../../vendor)
target_link_libraries(indago_emulation_worker PRIVATE unicorn)
install(TARGETS indago_emulation_worker RUNTIME DESTINATION emulation)
install(FILES ${unicorn_SOURCE_DIR}/COPYING ${unicorn_SOURCE_DIR}/COPYING.LGPL2 DESTINATION emulation/licenses)
install(FILES PROVENANCE.md DESTINATION emulation)
