cmake_minimum_required(VERSION 3.25)
project(IndagoDynamoRIO LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_C_STANDARD 11)
find_package(DynamoRIO REQUIRED)
if(UNIX)
    # Upstream's build-tree map assumes RelWithDebInfo even for a Release build.
    foreach(_target dynamorio dynamorio_static drinjectlib drdecode drlibc drmemfuncs drconfiglib drfrontendlib)
        if(TARGET ${_target})
            get_target_property(_configs ${_target} IMPORTED_CONFIGURATIONS)
            if("RELEASE" IN_LIST _configs)
                set_property(TARGET ${_target} PROPERTY MAP_IMPORTED_CONFIG_RELEASE Release)
            endif()
        endif()
    endforeach()
endif()
add_library(indago_dr_client SHARED client.c)
configure_DynamoRIO_client(indago_dr_client)
add_executable(indago_dr_host host.c)
target_include_directories(indago_dr_host PRIVATE "${DynamoRIO_INCLUDE_DIRS}")
target_link_libraries(indago_dr_host PRIVATE drinjectlib drconfiglib)
if(WIN32)
    target_compile_definitions(indago_dr_host PRIVATE WINDOWS _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_definitions(indago_dr_host PRIVATE LINUX)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    target_compile_definitions(indago_dr_host PRIVATE X86_64)
else()
    target_compile_definitions(indago_dr_host PRIVATE X86_32)
endif()
